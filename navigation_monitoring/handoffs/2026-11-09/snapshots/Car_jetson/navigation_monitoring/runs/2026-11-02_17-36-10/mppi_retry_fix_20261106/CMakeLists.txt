cmake_minimum_required(VERSION 3.16)
project(mppi_retry_offline_validation LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(xtensor REQUIRED)
find_package(xsimd REQUIRED)
get_filename_component(CAR_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(MPPI_RELATIVE "nav2/src/navigation2/nav2_mppi_controller")
set(MPPI_SOURCE "${CAR_PROJECT}/${MPPI_RELATIVE}")
set(MPPI_BEFORE "${CMAKE_CURRENT_SOURCE_DIR}/before/${MPPI_RELATIVE}")

foreach(version IN ITEMS before after)
  if(version STREQUAL "before")
    set(OPTIMIZER_SOURCE "${MPPI_BEFORE}/src/optimizer.cpp")
    set(OPTIMIZER_INCLUDE "${MPPI_BEFORE}/include")
    set(REGRESSION_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/retry_harness.cpp")
  else()
    set(OPTIMIZER_SOURCE "${MPPI_SOURCE}/src/optimizer.cpp")
    set(OPTIMIZER_INCLUDE "${MPPI_SOURCE}/include")
    set(REGRESSION_SOURCE "${MPPI_SOURCE}/test/optimizer_retry_regression_test.cpp")
  endif()
  add_executable(retry_${version}
    "${REGRESSION_SOURCE}" "${OPTIMIZER_SOURCE}" "${MPPI_SOURCE}/src/critic_manager.cpp")
  target_compile_features(retry_${version} PRIVATE cxx_std_17)
  target_compile_options(retry_${version} PRIVATE -fconcepts)
  target_compile_definitions(retry_${version} PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
  if(version STREQUAL "before")
    target_compile_definitions(retry_${version} PRIVATE LEGACY_FALLBACK_SIGNATURE)
  endif()
  target_include_directories(retry_${version} BEFORE PRIVATE "${OPTIMIZER_INCLUDE}" "${MPPI_SOURCE}/include")
  ament_target_dependencies(retry_${version} nav2_mppi_controller)
  target_link_libraries(retry_${version} xtensor xtensor::optimize xtensor::use_xsimd)
endforeach()
